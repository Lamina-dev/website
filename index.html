// 精确数学计算
print("Hello, Lamina!");
var fraction = 16 / 9; // 精确分数，不是浮点数
print("16/9 =", fraction); // 输出: 16/9

var root = sqrt(2); // 精确无理数
print("√2 =", root); // 输出: √2
print("√2 × √2 =", root * root);// 输出: 2（精确结果）

// 向量运算
var v1 = [1, 2, 3];
var v2 = [4, 5, 6];
print("v1 + v2 =", v1 + v2);
print("v1 · v2 =", dot(v1, v2));
